2.1 Autoload单例清单

    在Godot中，Autoload是全局可访问的单例节点，在项目设置中配置。
    所有单例在游戏启动时自动加载，提供全局数据和服务。

| 单例名称 | 脚本文件 | 职责 |
|---------|---------|------|
| GameState | game_state.gd | 核心全局状态：时间、世界参数、当前模式 |
| MaterialRegistry | material_registry.gd | 加载和管理所有MaterialData资源 |
| Registry | registry.gd | 模组注册表，存储所有模组添加的物品/矿物/植物等 |
| ModManager | mod_manager.gd | 模组加载、校验、版本管理 |
| SaveManager | save_manager.gd | 存档、读档、自动存档 |
| EcosystemManager | ecosystem_manager.gd | 生态数据：物种数量、食物链平衡 |
| EventBus | event_bus.gd | 全局事件总线，用于解耦模块间通信 |

2.2 GameState单例（核心全局状态）

    GameState是全局状态中心，所有需要跨场景、跨模块共享的数据都存储于此。

    ```gdscript
    # game_state.gd (Autoload)
    extends Node
    
    # 游戏版本
    var game_version: String = "1.0.0"
    
    # 世界设置
    var world_size: int = 80
    var world_seed: int = 42
    var terrain_types: Array[String] = ["plains", "forest", "hills", "mountains", "water", "swamp", "tundra", "volcanic"]
    var food_count: int = 30
    var predator_count: int = 3
    
    # 时间系统
    var time_ratio: float = 1.0          # 游戏时间/现实时间
    var simulation_speed: float = 1.0     # 独立模拟倍率
    var game_time: float = 0.0            # 游戏总秒数
    var is_pure_compute_mode: bool = false
    
    const DAY_LENGTH: float = 600.0       # 游戏秒/天（默认10分钟）
    const YEAR_LENGTH: float = 60000.0    # 游戏秒/年（默认100分钟）
    
    # 当前模式
    enum ObservationMode { SPECTATOR, AVATAR, FOLLOW }
    var current_mode: ObservationMode = ObservationMode.SPECTATOR
    
    # 当前选中的AHEI（用于面板显示）
    var selected_ahei: Node = null
    
    # 图形与界面
    var graphics_quality: String = "medium"
    var ui_language: String = "chinese"
    var mouse_sensitivity: float = 0.75
    var invert_y: bool = false
    
    # 时间工具方法
    func get_current_day() -> int:
        return int(game_time / DAY_LENGTH)
    
    func get_current_season() -> String:
        var year_progress = fmod(game_time, YEAR_LENGTH) / YEAR_LENGTH
        if year_progress < 0.25: return "spring"
        elif year_progress < 0.5: return "summer"
        elif year_progress < 0.75: return "autumn"
        else: return "winter"
    
    func get_effective_time_multiplier() -> float:
        return time_ratio * simulation_speed
    ```

2.3 MaterialRegistry单例

    ```gdscript
    # material_registry.gd (Autoload)
    extends Node
    
    var materials: Dictionary = {}  # {material_id: MaterialData}
    
    func _ready() -> void:
        _load_all_materials()
    
    func _load_all_materials() -> void:
        var dir = DirAccess.open("res://resources/materials/")
        if not dir: return
        dir.list_dir_begin()
        var file_name = dir.get_next()
        while file_name != "":
            if file_name.ends_with(".tres"):
                var mat = load("res://resources/materials/" + file_name) as MaterialData
                if mat:
                    materials[mat.material_id] = mat
            file_name = dir.get_next()
    
    func get_material(id: String) -> MaterialData:
        return materials.get(id, null)
    ```

2.6 SaveManager单例

    ```gdscript
    # save_manager.gd (Autoload)
    extends Node
    
    const SAVE_DIR = "user://saves/"
    
    func save_game(save_name: String) -> bool:
        DirAccess.make_dir_absolute(SAVE_DIR)
        var save_data = {
            "version": GameState.game_version,
            "world_seed": GameState.world_seed,
            "game_time": GameState.game_time,
            "time_ratio": GameState.time_ratio,
            # ... 其他需要保存的状态
        }
        var file = FileAccess.open(SAVE_DIR + save_name + ".sav", FileAccess.WRITE)
        if not file: return false
        file.store_string(JSON.stringify(save_data))
        return true
    
    func load_game(save_name: String) -> Dictionary:
        var file = FileAccess.open(SAVE_DIR + save_name + ".sav", FileAccess.READ)
        if not file: return {}
        return JSON.parse_string(file.get_as_text())
    ```

2.7 EcosystemManager单例

    ```gdscript
    # ecosystem_manager.gd (Autoload)
    extends Node
    
    var plant_count: int = 0
    var herbivore_count: int = 0
    var predator_count: int = 0
    var ahei_count: int = 0
    
    func register_entity(type: String) -> void:
        match type:
            "plant": plant_count += 1
            "herbivore": herbivore_count += 1
            "predator": predator_count += 1
            "ahei": ahei_count += 1
    
    func unregister_entity(type: String) -> void:
        match type:
            "plant": plant_count = max(0, plant_count - 1)
            "herbivore": herbivore_count = max(0, herbivore_count - 1)
            "predator": predator_count = max(0, predator_count - 1)
            "ahei": ahei_count = max(0, ahei_count - 1)
    ```

4.1 主入口（main.gd）

    挂载到主场景Main节点上。
    ```gdscript
    # main.gd
    extends Node3D
    
    @onready var world_manager: WorldManager = $World
    @onready var spectator_camera: SpectatorCamera = $Cameras/SpectatorCamera
    @onready var player: PlayerController = $Player
    
    func _ready() -> void:
        # 初始化世界
        world_manager.generate_world()
        # 设置初始摄像机
        _set_mode(GameState.ObservationMode.SPECTATOR)
    
    func _process(delta: float) -> void:
        # 时间更新
        GameState.game_time += delta * GameState.get_effective_time_multiplier()
        # 天气更新
        $Weather.update_weather(delta)
    
    func _input(event: InputEvent) -> void:
        if event.is_action_pressed("switch_avatar"):
            _toggle_avatar_mode()
        elif event.is_action_pressed("open_panel"):
            $GlobalPanel.toggle()
        elif event.is_action_pressed("pause"):
            $PauseMenu.toggle()
    
    func _toggle_avatar_mode() -> void:
        if GameState.current_mode == GameState.ObservationMode.AVATAR:
            _set_mode(GameState.ObservationMode.SPECTATOR)
        else:
            _set_mode(GameState.ObservationMode.AVATAR)
    
    func _set_mode(mode: int) -> void:
        GameState.current_mode = mode
        match mode:
            GameState.ObservationMode.SPECTATOR:
                player.hide()
                spectator_camera.make_current()
                Input.set_mouse_mode(Input.MOUSE_MODE_CAPTURED)
            GameState.ObservationMode.AVATAR:
                player.show()
                player.activate()
                Input.set_mouse_mode(Input.MOUSE_MODE_CAPTURED)
    ```

4.3 角色控制器（player_controller.gd — 降临模式）

    ```gdscript
    # player_controller.gd
    class_name PlayerController
    extends CharacterBody3D
    
    @export var move_speed: float = 8.0
    @export var jump_velocity: float = 6.0
    @export var mouse_sensitivity: float = 0.3
    
    var gravity: float = ProjectSettings.get_setting("physics/3d/default_gravity")
    var held_item: Node3D = null
    var held_item_name: String = ""
    
    @onready var camera_mount: Node3D = $CameraMount
    @onready var spring_arm: SpringArm3D = $SpringArm3D
    @onready var camera_3d: Camera3D = $SpringArm3D/Camera3D
    
    func _ready() -> void:
        hide()
    
    func activate() -> void:
        show()
        camera_3d.make_current()
    
    func _physics_process(delta: float) -> void:
        if GameState.current_mode != GameState.ObservationMode.AVATAR:
            return
        
        # 重力
        if not is_on_floor():
            velocity.y -= gravity * delta
        
        # 移动输入
        var input_dir = Input.get_vector("move_left", "move_right", "move_forward", "move_backward")
        var direction = (transform.basis * Vector3(input_dir.x, 0, input_dir.y)).normalized()
        if direction:
            velocity.x = direction.x * move_speed
            velocity.z = direction.z * move_speed
        else:
            velocity.x = move_toward(velocity.x, 0, move_speed)
            velocity.z = move_toward(velocity.z, 0, move_speed)
        
        move_and_slide()
    
    func _input(event: InputEvent) -> void:
        if GameState.current_mode != GameState.ObservationMode.AVATAR:
            return
        
        if event is InputEventMouseMotion and Input.get_mouse_mode() == Input.MOUSE_MODE_CAPTURED:
            rotate_y(-event.relative.x * mouse_sensitivity * 0.01)
            camera_mount.rotate_x(-event.relative.y * mouse_sensitivity * 0.01)
            camera_mount.rotation.x = clampf(camera_mount.rotation.x, -PI/2, PI/2)
        
        if event.is_action_pressed("jump") and is_on_floor():
            velocity.y = jump_velocity
        
        if event.is_action_pressed("interact"):
            _on_interact()
        
        if event.is_action_pressed("drop_item"):
            _drop_item()
    
    func _on_interact() -> void:
        var space_state = get_world_3d().direct_space_state
        var query = PhysicsRayQueryParameters3D.create(camera_3d.global_position, camera_3d.global_position - camera_3d.global_transform.basis.z * 3.0)
        query.exclude = [self]
        var result = space_state.intersect_ray(query)
        if result:
            var target = result.collider
            if target.has_method("on_interact"):
                target.on_interact(self)
    
    func _drop_item() -> void:
        if held_item:
            held_item.reparent(get_tree().current_scene)
            held_item.global_position = global_position + Vector3.FORWARD * 1.5
            held_item = null
            held_item_name = ""
    ```

4.5 地形生成器（terrain_generator.gd）

    ```gdscript
    # terrain_generator.gd
    class_name TerrainGenerator
    extends RefCounted
    
    const CHUNK_SIZE: int = 50
    
    func generate_terrain_chunks(parent: Node3D) -> void:
        # 为整个WORLD_SIZE范围生成地形网格
        var surface_tool = SurfaceTool.new()
        surface_tool.begin(Mesh.PRIMITIVE_TRIANGLES)
        
        var size = GameState.world_size
        var step = 1.0
        var rows = int(size / step) + 1
        var cols = int(size / step) + 1
        
        # 生成顶点和颜色
        for z in range(-size/2, size/2 + step, step):
            for x in range(-size/2, size/2 + step, step):
                var y = _get_height(x, z)
                surface_tool.set_color(_get_terrain_color(y))
                surface_tool.add_vertex(Vector3(x, y, z))
        
        # 生成索引
        for r in range(rows - 1):
            for c in range(cols - 1):
                var i0 = r * cols + c
                var i1 = r * cols + c + 1
                var i2 = (r + 1) * cols + c
                var i3 = (r + 1) * cols + c + 1
                surface_tool.add_index(i0)
                surface_tool.add_index(i2)
                surface_tool.add_index(i1)
                surface_tool.add_index(i1)
                surface_tool.add_index(i2)
                surface_tool.add_index(i3)
        
        surface_tool.generate_normals()
        var mesh = surface_tool.commit()
        
        var mesh_instance = MeshInstance3D.new()
        mesh_instance.mesh = mesh
        
        # 设置材质（顶点颜色着色器）
        var mat = StandardMaterial3D.new()
        mat.vertex_color_use_as_albedo = true
        mesh_instance.set_surface_override_material(0, mat)
        
        # 添加碰撞体
        var static_body = StaticBody3D.new()
        var collision_shape = CollisionShape3D.new()
        var shape = ConcavePolygonShape3D.new()
        shape.set_faces(mesh.get_faces())
        collision_shape.shape = shape
        static_body.add_child(collision_shape)
        mesh_instance.add_child(static_body)
        
        parent.add_child(mesh_instance)
    
    func _get_height(x: float, z: float) -> float:
        # 使用NoiseUtil计算高度（与文档5一致）
        var h1 = NoiseUtil.perlin_noise(x * 0.004, z * 0.004, 1) * 28.0
        var h2 = NoiseUtil.perlin_noise(x * 0.02, z * 0.02, 2) * 10.0
        var h3 = NoiseUtil.perlin_noise(x * 0.08, z * 0.08, 3) * 4.0
        return h1 + h2 + h3
    
    func _get_terrain_color(y: float) -> Color:
        if y < -1: return Color(0.71, 0.67, 0.47)    # 沙地
        elif y < 2: return Color(0.75, 0.71, 0.55)    # 沙滩
        elif y < 8: return Color(0.20, 0.59, 0.16)    # 草地
        elif y < 18: return Color(0.27, 0.51, 0.18)   # 丘陵绿
        elif y < 30: return Color(0.39, 0.43, 0.31)   # 岩石混合
        elif y < 50: return Color(0.55, 0.55, 0.55)   # 山地灰
        elif y < 65: return Color(0.78, 0.78, 0.82)   # 高山
        else: return Color(0.96, 0.96, 0.98)           # 雪线
    ```

    ```gdscript
    # genome.gd
    class_name Genome
    extends Resource
    
    @export var sex_chromosomes: String = "XX"  # "XX" 或 "XY"
    @export var trait_genes: Dictionary = {
        "body_size": 1.0,
        "color_hue": 0.12,
        "color_saturation": 0.5,
        "symmetry": 0.9,
        "curiosity": 0.5,
        "aggression": 0.5,
        "mirror_sensitivity": 0.5,
        # ... 全部30+可遗传性状
    }
    
    func get_sex() -> String:
        return "female" if sex_chromosomes == "XX" else "male"
    
    static func recombine(male_genome: Genome, female_genome: Genome) -> Genome:
        var child = Genome.new()
        child.sex_chromosomes = "X" + ("X" if randi() % 2 == 0 else "Y")
        for trait in child.trait_genes:
            if randf() < 0.5:
                child.trait_genes[trait] = male_genome.trait_genes[trait]
            else:
                child.trait_genes[trait] = female_genome.trait_genes[trait]
            child.trait_genes[trait] += randf_range(-0.02, 0.02)
        return child
    ```

第七章   常见问题（FAQ）
================================================================================

Q：从Ursina换到Godot，哪些代码可以复用？
A：所有的设计文档和算法逻辑都可以复用。需要重写的是API调用和节点创建部分。
   Python的预测编码网络（PyTorch）完全保留，通过WebSocket与Godot通信。

Q：Godot项目中的Python代码放在哪里？
A：Python部分独立运行（PyTorch + WebSocket服务器），不在Godot项目内。
   Godot通过WebSocketClient连接Python服务器，获取AI计算结果。

Q：地形生成与Ursina版本有差异吗？
A：算法完全一致（多层正弦波叠加），只是网格构建API不同
   (SurfaceTool vs Ursina Mesh)。

Q：模组系统在Godot中如何工作？
A：模组放在mods/文件夹，包含mod_info.json和init.gd。
   ModManager在启动时扫描所有模组，调用init.gd中的register(api)函数。

Q：如何切换模型风格（几何体/二次元/写实）？
A：在M键模型选择面板中切换。系统通过SkeletonMapper自动映射骨骼，
   装备自动挂载到正确位置。切换后会有短暂的适应期。

Q：性能在Godot中如何？
A：Godot 4.x的Vulkan渲染器性能远超Ursina/Panda3D。
   100个简单模型的AHEI + 地形 + 植被在RTX 5060上可稳定60FPS+。

Q：分阶段开发中，每个阶段的验证标准是什么？
A：每个阶段完成后必须满足：代码无报错、核心功能可手动测试、
   输出日志符合预期、帧率在预算范围内。
